cmake_minimum_required(VERSION 3.20)
project(FlexModelCompileFail LANGUAGES CXX)
add_library(negative_case OBJECT "${CASE_SOURCE}")
target_include_directories(negative_case PRIVATE "${INCLUDE_DIR}")
target_compile_features(negative_case PRIVATE cxx_std_20)
set_target_properties(negative_case PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(negative_case PRIVATE /permissive- /utf-8)
endif()
